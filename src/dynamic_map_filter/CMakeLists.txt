cmake_minimum_required(VERSION 3.17)
project(dynamic_map_filter)
find_package(ament_cmake REQUIRED)
find_package(TBB REQUIRED)
find_package(OpenMP REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
find_package(OpenCV REQUIRED COMPONENTS core imgproc photo imgcodecs)
add_library(ptr_dynamic_map STATIC src/dufomap_filter.cpp)
set_target_properties(ptr_dynamic_map PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
target_include_directories(ptr_dynamic_map PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(ptr_dynamic_map SYSTEM PRIVATE vendor/ufomap/include vendor/include)
target_compile_definitions(ptr_dynamic_map PRIVATE UFO_PARALLEL=1)
target_link_libraries(ptr_dynamic_map PRIVATE TBB::tbb OpenMP::OpenMP_CXX lz4)
target_compile_options(ptr_dynamic_map PRIVATE -O3)
add_library(ptr_freedom STATIC src/freedom_filter.cpp
  vendor/freedom/src/freedom.cpp vendor/freedom/src/map.cpp vendor/freedom/src/mrmap.cpp
  vendor/freedom/src/scanmap.cpp vendor/freedom/src/raycast.cpp vendor/freedom/src/depth_image.cpp)
set_target_properties(ptr_freedom PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_include_directories(ptr_freedom PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(ptr_freedom SYSTEM PRIVATE vendor/freedom/include ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ptr_freedom PRIVATE ${PCL_LIBRARIES} ${OpenCV_LIBRARIES})
target_compile_options(ptr_freedom PRIVATE -O3)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_dufomap_filter test/test_dufomap_filter.cpp)
  target_link_libraries(test_dufomap_filter ptr_dynamic_map)
  ament_add_gtest(test_freedom_filter test/test_freedom_filter.cpp)
  target_link_libraries(test_freedom_filter ptr_freedom)
  target_include_directories(test_freedom_filter SYSTEM PRIVATE vendor/freedom/include ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
  add_executable(replay_filter test/replay_filter.cpp)
  target_link_libraries(replay_filter ptr_dynamic_map ptr_freedom)
endif()
install(TARGETS ptr_dynamic_map ptr_freedom EXPORT export_ptr_dynamic_map ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY vendor/ DESTINATION share/${PROJECT_NAME}/vendor)
ament_export_targets(export_ptr_dynamic_map HAS_LIBRARY_TARGET)
ament_export_dependencies(TBB OpenMP PCL OpenCV)
ament_package()
